# 第一部分：存储机制升级

## 1.1 现状问题

当前存储方案是纯 JSONL 文件 + JSON 侧车：

```
~/.my-agent/sessions/
├── session-abc123.jsonl           ← 消息体（append-only）
├── session-abc123.context.json    ← 侧车（plan/账本/轮次）
├── session-def456.jsonl
└── session-def456.context.json
```

| 问题 | 影响 |
|------|------|
| 会话列表需扫描目录 + 逐文件 parse | 1000 个会话时极慢 |
| 无法按模型/时间/用量搜索或排序 | 管理困难 |
| 无用量统计 | 不知道花了多少 token |
| 配置在 `config.json`，不跟随 session | 切换配置需手动编辑文件 |
| Skill 信息只存内存，重启丢失启用状态 | 每次启动重新扫描 |
| 无数据迁移/备份机制 | 升级困难 |

## 1.2 目标存储架构

对标 Orkas 的数据分层设计：

```
~/.my-agent/
```

```
SQLite  ←  索引、查询、统计（元数据）
JSONL   ←  消息体（append-only，高性能写入）
文件系统 ←  Skill 源文件、超大工具输出、日志
```

```ts
// src/storage/db.ts
import Database from "better-sqlite3";
import * as path from "node:path";
import * as fs from "node:fs";

let _db: Database.Database | null = null;

/** 获取数据库实例（惰性初始化） */
export function getDb(): Database.Database {
  if (_db) return _db;
  const dir = dataDir();
  fs.mkdirSync(dir, { recursive: true });
  const dbPath = path.join(dir, "my-agent.db");
  _db = new Database(dbPath);
  _db.pragma("journal_mode = WAL");       // 写性能 + 并发读
  _db.pragma("foreign_keys = ON");
  migrate(_db);
  return _db;
}

/** 关闭数据库（应用退出时调用） */
export function closeDb(): void {
  if (_db) { _db.close(); _db = null; }
}
```

```ts
        dir         TEXT NOT NULL,                 -- SKILL.md 所在目录
        enabled     INTEGER NOT NULL DEFAULT 1,
        installed_at INTEGER NOT NULL,
        updated_at  INTEGER NOT NULL
      );

      -- Schema 版本
      CREATE TABLE IF NOT EXISTS schema_version (
        version INTEGER PRIMARY KEY
      );
    `,
  },
];

function migrate(db: Database.Database): void {
  db.exec("CREATE TABLE IF NOT EXISTS schema_version (version INTEGER PRIMARY KEY)");
  const row = db.prepare("SELECT MAX(version) as v FROM schema_version").get() as
    { v: number | null };
  const current = row?.v ?? 0;

  for (const m of MIGRATIONS) {
    if (m.version > current) {
      db.exec(m.sql);
      db.prepare("INSERT OR REPLACE INTO schema_version (version) VALUES (?)").run(m.version);
    }
  }
}
```

### 1.3.4 会话数据访问层

```ts
// src/storage/session-repo.ts
import { getDb } from "./db.js";

export interface SessionMeta {
  id: string;
  name: string;
  model: string;
  provider: string;
  messageCount: number;
  inputTokens: number;
  outputTokens: number;
  createdAt: number;
  updatedAt: number;
  isArchived: boolean;
}

/** 创建或更新会话元数据 */
export function upsertSession(meta: Omit<SessionMeta, "isArchived">): void {
  const db = getDb();
  db.prepare(`
    INSERT INTO sessions (id, name, model, provider, message_count,
      input_tokens, output_tokens, created_at, updated_at)
    VALUES (@id, @name, @model, @provider, @messageCount,
      @inputTokens, @outputTokens, @createdAt, @updatedAt)
    ON CONFLICT(id) DO UPDATE SET
      name = excluded.name,
      model = excluded.model,
      message_count = excluded.message_count,
      input_tokens = excluded.input_tokens,
      output_tokens = excluded.output_tokens,
      updated_at = excluded.updated_at
  `).run(meta);
}

/** 列出会话（支持搜索和分页） */
export function listSessions(opts?: {
  search?: string;
  offset?: number;
  limit?: number;
  archived?: boolean;
}): SessionMeta[] {
  const db = getDb();
  const conditions: string[] = [];
  const params: Record<string, unknown> = {};

  if (opts?.search) {
    conditions.push("name LIKE @search");
    params.search = `%${opts.search}%`;
  }
  if (opts?.archived === false) {
    conditions.push("is_archived = 0");
  }

  const where = conditions.length > 0 ? `WHERE ${conditions.join(" AND ")}` : "";
  const limit = opts?.limit ?? 50;
  const offset = opts?.offset ?? 0;

  return db.prepare(`
    SELECT id, name, model, provider,
           message_count AS messageCount,
           input_tokens AS inputTokens,
           output_tokens AS outputTokens,
           created_at AS createdAt,
           updated_at AS updatedAt,
           is_archived AS isArchived
    FROM sessions
    ${where}
    ORDER BY updated_at DESC
    LIMIT @limit OFFSET @offset
  `).all({ ...params, limit, offset }) as SessionMeta[];
}

/** 删除会话元数据（级联删除 usage_logs） */
export function deleteSession(sessionId: string): void {
  const db = getDb();
  db.prepare("DELETE FROM sessions WHERE id = ?").run(sessionId);
}
```

### 1.3.5 用量统计

```ts
// src/storage/usage-repo.ts
import { getDb } from "./db.js";
import type { Usage } from "../shared/types.js";

export function logUsage(input: {
  sessionId: string;
  model: string;
  provider: string;
  usage: Usage;
  toolLoops: number;
  durationMs: number;
}): void {
  const db = getDb();
  db.prepare(`
    INSERT INTO usage_logs (session_id, model, provider,
      input_tokens, output_tokens, cache_read_tokens, cache_write_tokens,
      tool_loops, duration_ms, created_at)
    VALUES (@sessionId, @model, @provider,
      @inputTokens, @outputTokens, @cacheRead, @cacheWrite,
      @toolLoops, @durationMs, @createdAt)
  `).run({
    sessionId: input.sessionId,
    model: input.model,
    provider: input.provider,
    inputTokens: input.usage.inputTokens,
    outputTokens: input.usage.outputTokens,
    cacheRead: input.usage.cacheReadTokens ?? 0,
    cacheWrite: input.usage.cacheWriteTokens ?? 0,
    toolLoops: input.toolLoops,
    durationMs: input.durationMs,
    createdAt: Date.now(),
  });

  // 同步更新 sessions 表的聚合值
  db.prepare(`
    UPDATE sessions
    SET input_tokens = input_tokens + @inputTokens,
        output_tokens = output_tokens + @outputTokens
    WHERE id = @sessionId
  `).run({
    inputTokens: input.usage.inputTokens,
    outputTokens: input.usage.outputTokens,
    sessionId: input.sessionId,
  });
}

/** 按天聚合 token 用量 */
export function dailyUsage(days = 30): Array<{
  date: string;
  inputTokens: number;
  outputTokens: number;
}> {
  const db = getDb();
  return db.prepare(`
    SELECT date(created_at / 1000, 'unixepoch') AS date,
           SUM(input_tokens) AS inputTokens,
           SUM(output_tokens) AS outputTokens
    FROM usage_logs
    WHERE created_at > @since
    GROUP BY date
    ORDER BY date DESC
  `).all({ since: Date.now() - days * 86400_000 }) as any[];
}
```

## 1.5 文件锁机制

防止多实例并发写入同一 Session 文件。

```ts
// src/storage/locks.ts
import * as fs from "node:fs";
import * as path from "node:path";

const LOCK_DIR = path.join(dataDir(), "locks");

/**
 * 基于 lockfile 的简单互斥锁。
 * 对标 Orkas util/locks.ts 的文件级锁。
 */
export class FileLock {
  private lockPath: string;
  private acquired = false;

  constructor(name: string) {
    fs.mkdirSync(LOCK_DIR, { recursive: true });
    this.lockPath = path.join(LOCK_DIR, `${name}.lock`);
  }

  /** 尝试获取锁，超时返回 false */
  acquire(timeoutMs = 5000): boolean {
    const deadline = Date.now() + timeoutMs;
    while (Date.now() < deadline) {
      try {
        fs.writeFileSync(this.lockPath, String(process.pid), { flag: "wx" });
        this.acquired = true;
        return true;
      } catch (err) {
        if ((err as NodeJS.ErrnoException).code !== "EEXIST") throw err;
      }
      // 锁被持有 → 检查持有者是否还活着
      try {
        const pid = Number.parseInt(fs.readFileSync(this.lockPath, "utf-8"), 10);
        if (!isProcessAlive(pid)) {
          // 僵尸锁 → 清理
          fs.unlinkSync(this.lockPath);
          continue;
        }
      } catch { /* 读取失败，继续等待 */ }
    }
    return false;
  }

  release(): void {
    if (!this.acquired) return;
    try { fs.unlinkSync(this.lockPath); } catch { /* 忽略 */ }
    this.acquired = false;
  }
}

function isProcessAlive(pid: number): boolean {
  try {
    process.kill(pid, 0);
    return true;
  } catch {
    return false;
  }
}
```

### 2.3.3 Preload

```js
// electron/preload.js
// ⚠️ 必须是 .js，不跑 tsx hook
const { contextBridge, ipcRenderer } = require("electron");

contextBridge.exposeInMainWorld("myAgent", {
  // ============================================================
  // 请求-响应（invoke/handle 模式）
  // ============================================================
  invoke: (channel, ...args) => ipcRenderer.invoke(channel, ...args),

  // ============================================================
  // 流式通道（用于 Agent SSE 输出 + 工具执行事件）
  // ============================================================
  stream: (channel, payload) => {
    const streamId = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
    ipcRenderer.send(channel, { streamId, ...payload });

    return {
      // 监听特定类型的事件
      on: (event, callback) => {
        const listener = (_ev, data) => {
          if (data.streamId === streamId) callback(data.payload);
        };
        ipcRenderer.on(`stream:${event}`, listener);
        return () => ipcRenderer.removeListener(`stream:${event}`, listener);
      },
      // 取消流
      cancel: () => {
        ipcRenderer.send(`${channel}:cancel`, { streamId });
      },
    };
  },

  // ============================================================
  // 主进程 → Renderer 的事件推送
  // ============================================================
  on: (channel, callback) => {
    const listener = (_ev, ...args) => callback(...args);
    ipcRenderer.on(channel, listener);
    return () => ipcRenderer.removeListener(channel, listener);
  },
});
```

```ts
import { AgentRunner } from "../../agent/runner.js";
import { SessionStore } from "../../storage/session-store.js";
import { ProviderRegistry } from "../../providers/registry.js";
import { BUILTIN_TOOLS } from "../../tools/builtin.js";
import { loadConfig } from "../../config/loader.js";
import { buildSystemPrompt } from "../../prompts/system-prompt-builder.js";
import { logUsage } from "../../storage/usage-repo.js";
import { upsertSession } from "../../storage/session-repo.js";
import type { AgentRunEvent } from "../../agent/types.js";

// 保持活跃的 runner 引用，用于取消
const activeRunners = new Map<string, AgentRunner>();

export async function* streamChat(input: {
  message: string;
  sessionId?: string;
  model?: string;
  workingDir?: string;
}): AsyncIterable<AgentRunEvent & { sessionId?: string }> {
  const config = await loadConfig();
  const providers = new ProviderRegistry(config);
  const store = new SessionStore();

  // 创建或恢复 session
  const session = input.sessionId
    ? (store.get(input.sessionId) ?? store.create())
    : store.create();

  // 创建 runner
  const runner = new AgentRunner({
    config,
    providers,
    tools: BUILTIN_TOOLS,
    session,
  });
  activeRunners.set(session.sessionId, runner);

  const systemPrompt = buildSystemPrompt({ /* ... */ }).systemPrompt;

  let lastUsage = { inputTokens: 0, outputTokens: 0, cacheReadTokens: 0, cacheWriteTokens: 0, totalTokens: 0 };
  let toolLoops = 0;

  try {
    for await (const ev of runner.runStream({
      message: input.message,
      model: input.model,
      systemPrompt,
      workingDir: input.workingDir,
    })) {
      // 注入 sessionId 让前端知道当前会话
      yield { ...ev, sessionId: session.sessionId };

      if (ev.type === "done") {
        lastUsage = ev.result.meta.usage;
        toolLoops = ev.result.meta.toolLoops;

        // 写入用量日志
        logUsage({
          sessionId: session.sessionId,
          model: ev.result.meta.model,
          provider: ev.result.meta.provider,
          usage: lastUsage,
          toolLoops,
          durationMs: ev.result.meta.durationMs,
        });

        // 更新会话元数据
        upsertSession({
          id: session.sessionId,
          name: session.getDisplayName(),
          model: ev.result.meta.model,
          provider: ev.result.meta.provider,
          messageCount: 0,  // 由 Session 计算
          inputTokens: lastUsage.inputTokens,
          outputTokens: lastUsage.outputTokens,
          createdAt: Date.now(),
          updatedAt: Date.now(),
        });
      }
    }
  } finally {
    activeRunners.delete(session.sessionId);
  }
}

export function abortChat(sessionId: string): void {
  activeRunners.delete(sessionId);
  // Runner 会通过 AbortSignal 处理取消
}
```

```js
function renderMarkdown(text) {
  if (typeof marked === "undefined") {
    // fallback: 简单的转义 + 换行
    return escapeHtml(text).replace(/\n/g, "<br>");
  }

  return marked.parse(text, {
    breaks: true,
    gfm: true,
  });
}

function escapeHtml(str) {
  const map = { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" };
  return str.replace(/[&<>"']/g, (c) => map[c]);
}
```

```ts
private async *prepareContextBeforeModelCall(
  provider: LLMProvider,
  modelId: string,
  cacheRetention: string | undefined,
  control: CompactionControl,
  recordUsage: (u: Usage) => void,
  incrementCompactionCount: () => void,
): AsyncIterable<AgentRunEvent> {
  // 检查上下文是否超出 82%
  const modelConfig = this.config.models.catalog[modelId];
  const contextWindow = modelConfig?.contextWindow ?? 128_000;
  const estimatedTokens = this.session.estimateModelTokens();

  if (estimatedTokens < contextWindow * CONTEXT_COMPACTION_TRIGGER_RATIO) {
    return; // 未触发
  }

  // 检查压缩预算
  if (control.epochs >= control.maxEpochs) {
    if (!control.limitLogged) {
      control.limitLogged = true;
      // log.warn("compaction budget exhausted")
    }
    return;
  }

  // === 第一层：历史摘要 ===
  const historyCandidate = this.session.getPendingHistoryArchive();
  if (historyCandidate && historyCandidate.turnIds.length > 0) {
    const fingerprint = `history:${historyCandidate.turnIds.join(",")}:${historyCandidate.rawTokens}`;
    if (control.attemptedFingerprints.has(fingerprint)) return; // 防抖

    control.attemptedFingerprints.add(fingerprint);
    control.attempts++;

    yield { type: "context_status", phase: "history_summary_start",
      message: `Compacting ${historyCandidate.turnIds.length} turns...` };

    try {
      const summary = await this.generateCompactionSummary(
        provider, modelId, historyCandidate, cacheRetention,
      );

      // 验证节省量
      const tokensAfter = this.session.estimateModelTokens();
      const savings = historyCandidate.rawTokens - tokensAfter;
      const minSavings = Math.max(64, Math.min(6000, historyCandidate.rawTokens * 0.1));

      if (savings >= minSavings) {
        this.session.applyHistorySummary(summary, historyCandidate.turnIds);
        control.epochs++;
        incrementCompactionCount();
        yield { type: "compaction",
          tokensBefore: historyCandidate.rawTokens,

        message: "Compaction failed" };
      control.failures++;
    }
  }

  cacheRetention?: string,
```

```ts
import { getDb } from "../../storage/db.js";
import { skillsDir, builtinSkillsDir } from "../../storage/paths.js";
import type { SkillSpec, SkillContent } from "../../skills/types.js";

let _cache: SkillContent[] | null = null;

/** 获取所有 Skill（含启用状态），带缓存 */
export function list(): Array<SkillSpec & { enabled: boolean }> {
  const db = getDb();
  const allSpecs = [
    ...SkillLoader.scan(builtinSkillsDir(), "system"),
    ...SkillLoader.scan(skillsDir(), "user"),
  ];

  // 合并 DB 中的启用状态
  return allSpecs.map((spec) => {
    const row = db.prepare("SELECT enabled FROM skills_index WHERE id = ?")
      .get(spec.id) as { enabled: number } | undefined;
    return { ...spec, enabled: row?.enabled !== 0 };
  });
}

/** 获取单个 Skill 的完整内容 */
export function get(id: string): (SkillContent & { enabled: boolean }) | null {
  const specs = list();
  const spec = specs.find((s) => s.id === id);
  if (!spec) return null;
  const content = SkillLoader.load(spec);
  if (!content) return null;
  return { ...content, enabled: spec.enabled };
}

/** 设置启用/禁用 */
export function setEnabled(id: string, enabled: boolean): void {
  const db = getDb();
  db.prepare(`
    INSERT INTO skills_index (id, name, description_zh, description_en, source, dir, enabled, installed_at, updated_at)
    VALUES (@id, '', '', '', 'user', '', @enabled, @now, @now)
    ON CONFLICT(id) DO UPDATE SET enabled = @enabled, updated_at = @now
  `).run({ id, enabled: enabled ? 1 : 0, now: Date.now() });
  _cache = null; // 失效缓存
}

/** 触发词匹配：检查用户消息是否匹配任何 Skill 的 trigger */
export function matchTriggers(message: string): SkillContent[] {
  return list()
    .filter((s) => s.enabled)
    .map((s) => SkillLoader.load(s))
    .filter((s): s is SkillContent => {
      if (!s) return false;
      // 在 SKILL.md frontmatter 或正文开头查找触发词
      const triggers = extractTriggers(s);
      return triggers.some((t) => message.toLowerCase().includes(t.toLowerCase()));
    });
}

function extractTriggers(skill: SkillContent): string[] {
  // 从 SKILL.md 中提取触发词（如 "TRIGGER: chart,graph,plot"）
  const match = skill.body.match(/TRIGGER[s]?:\s*(.+)/i);
  if (!match) return [];
  return match[1].split(",").map((s) => s.trim().toLowerCase());
}
```

```ts
// src/prompts/system-prompt-builder.ts 增强

export function buildSkillContext(input: {
  enabledSkillIds?: string[];
  matchedTriggers?: SkillContent[];
  allSkills: Array<SkillSpec & { enabled: boolean }>;
}): string {
  const parts: string[] = [];

  // 1. 触发匹配的 Skill（最高优先级，注入完整内容）
  if (input.matchedTriggers && input.matchedTriggers.length > 0) {
    parts.push("## Active Skills (triggered by this request)\n");
    for (const skill of input.matchedTriggers) {
      parts.push(`### ${skill.name}\n\n${skill.body}\n`);
    }
  }

  // 2. Skill 索引（让 Agent 知道有哪些可用 Skill）
  const enabled = input.allSkills.filter((s) => s.enabled);
  if (enabled.length > 0) {
    parts.push("## Available Skills\n");
    for (const skill of enabled) {
      const desc = skill.description_zh || skill.description_en;
      parts.push(`- **${skill.name}** (\`${skill.id}\`): ${desc}`);
    }
    parts.push("\nSkill 会根据用户消息中的关键词自动触发。");
  }

  return parts.join("\n");
}
```

```ts
 */
export class PathSandbox {
  private allowedRoots: string[];

  constructor(opts: { workingDir: string; extraRoots?: string[] }) {
    this.allowedRoots = [
      path.resolve(opts.workingDir),
      // 工具结果溢出目录
      // 附件目录
      ...(opts.extraRoots ?? []),
    ];
  }

  /** 检查路径是否允许访问 */
  isPathAllowed(targetPath: string): boolean {
    const resolved = path.resolve(targetPath);
    return this.allowedRoots.some((root) => {
      const rel = path.relative(root, resolved);
      return rel !== "" && !rel.startsWith("..") && !path.isAbsolute(rel);
    });
  }

  /** 将相对路径解析到工作目录（仅当结果路径在沙箱内时） */
  resolveSafely(targetPath: string): string | null {
    const resolved = path.resolve(targetPath);
    if (!this.isPathAllowed(resolved)) return null;
    return resolved;
  }

  /** 禁止的路径模式 */
  private static readonly FORBIDDEN_PATTERNS = [
    /\/\.env$/i,
    /\/\.git\/config$/i,
    /\/\.ssh\//i,
    /\/etc\/passwd$/i,
    /\/Windows\/System32\//i,
    /\/boot\//i,
  ];

  static isForbiddenPath(targetPath: string): boolean {
    return PathSandbox.FORBIDDEN_PATTERNS.some((p) => p.test(targetPath));
  }
}
```

**在工具中的使用：**

```ts

  const sandbox = new PathSandbox({ workingDir: ctx.workingDir ?? process.cwd() });

    return { content: `❌ 路径访问被拒绝: ${input.filePath}`, isError: true };
  }
```

## 3.4 工具结果溢出管理

对标 Orkas 的 `util/tool-result-cap.ts`。超大工具输出不应全量进入 LLM 上下文。

```ts
// src/util/tool-result-cap.ts
import * as fs from "node:fs";
import * as path from "node:path";
import { toolResultsDir } from "../storage/paths.js";

const MAX_INLINE_CHARS = 12_000;  // 超过此值溢出到磁盘

/**
 * 处理工具结果：超长内容写磁盘，返回摘要 + 引用。
 */
export function capToolResult(input: {
  sessionId: string;
  content: string;
  toolName: string;
}): { content: string; persistedRef?: string } {
  if (input.content.length <= MAX_INLINE_CHARS) {
    return { content: input.content };
  }

  // 写入磁盘
  const dir = toolResultsDir(input.sessionId);
  fs.mkdirSync(dir, { recursive: true });
  const ref = `${input.toolName}-${Date.now()}.json`;
  const filePath = path.join(dir, ref);
  fs.writeFileSync(filePath, input.content, "utf-8");

  // 返回摘要
  const truncated = input.content.slice(0, MAX_INLINE_CHARS);
  return {
    content: `${truncated}\n\n...(输出被截断。完整结果 ${input.content.length} 字符)\n[已保存至: ${ref}]\n[使用 read_file 工具读取完整内容]`,
    persistedRef: ref,
  };
}
```

## 3.5 manage_execution_plan 工具

Session 中已有完整的执行计划基础设施（`ensureExecutionPlanAnchor` / `updateExecutionPlan` / `recordCompletedWork`），但缺少让 LLM 与之交互的工具。

```ts
// src/tools/execution-plan-tool.ts
import { defineTool } from "./base.js";
import type { AgentRunner } from "../agent/runner.js";

/**
 * 创建 manage_execution_plan 工具。
 * 需要传入 runner 引用以访问 session。
 */
export function createExecutionPlanTool(getRunner: () => AgentRunner) {
  return defineTool({
    name: "manage_execution_plan",
    description:
      "管理执行计划。用于创建、更新任务步骤列表。每次取得实质性进展后应更新计划状态。",
    inputSchema: {
      type: "object",
      properties: {
        explanation: {
          type: "string",
          description: "本次更新的简要说明（最多 500 字符）",
        },
        plan: {
          type: "array",
          description: "步骤列表（最多 12 步）",
          items: {
            type: "object",
            properties: {
              step: { type: "string", description: "步骤描述（最多 180 字符）" },
              status: {
                type: "string",
                enum: ["pending", "in_progress", "completed", "blocked"],
              },
            },
            required: ["step", "status"],
          },
        },
      },
      required: ["plan"],
    },
    execute: async (input) => {
      const runner = getRunner();
      const session = runner.getSession();
      const plan = input.plan as Array<{ step: string; status: string }>;

      session.updateExecutionPlan({
        steps: plan.map((s) => ({
          step: s.step,
          status: s.status as "pending" | "in_progress" | "completed" | "blocked",
        })),
      });

      const current = session.getExecutionPlan();
      const summary = current!.steps
        .map((s, i) => `${i + 1}. [${s.status}] ${s.step}`)
        .join("\n");

      return {
        content: `✅ 执行计划已更新:\n${summary}`,
      };
    },
  });
}
```
